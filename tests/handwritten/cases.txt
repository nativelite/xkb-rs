xkb_keymap { xkb_keycodes { <ABCDE> = 10; <I129> = 11; <a.b-c_d> = 12; }; xkb_types { type "T" { modifiers = Shift; map[Shift] = 2; }; }; xkb_compat { }; xkb_symbols { key <ABCDE> { [ a, b ] }; key <a.b-c_d> { [ c ] }; }; };
=== hex and upper-case X
--- raw
xkb_keymap { xkb_keycodes { <A> = 0X0a; }; xkb_types { type "T" { modifiers = Shift; map[Shift] = 2; }; }; xkb_compat { }; xkb_symbols { key <A> { [ 0X61, b ] }; }; };
=== symbols without brackets around a single keysym
--- raw
xkb_keymap { xkb_keycodes { <A> = 10; }; xkb_types { type "T" { modifiers = Shift; map[Shift] = 2; }; }; xkb_compat { }; xkb_symbols { key <A> { symbols[Group1] = [ a ] }; }; };
=== modifier_map with a trailing comma

=== key body with a trailing comma
--- raw
xkb_keymap { xkb_keycodes { <A> = 10; }; xkb_types { type "T" { modifiers = Shift; map[Shift] = 2; }; }; xkb_compat { }; xkb_symbols { key <A> { [ a, b ], }; }; };
